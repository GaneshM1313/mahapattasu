  getAll:          (params) => api.get('/sales', { params }),

  create:          (data)   => api.post('/sales', data),
  getSummary:      (params) => api.get('/sales/summary', { params }),
  downloadInvoice: (id)     => api.get(`/sales/${id}/invoice`, { responseType: 'blob' }),
